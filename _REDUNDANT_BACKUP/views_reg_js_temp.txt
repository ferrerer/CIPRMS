      function approveRecord(id) {
        const p = partnerships.find(x => x.id === id);

        p.status = 'Active';

      function confirmDelete() {
        if (!deletingId) return;
        const idx = partnerships.findIndex(x => x.id === deletingId);
        if (idx === -1) return;
        const name = partnerships[idx].inst;

        showToast(`"${name}" has been removed from the registry.`);
      }

      function computeRenewStatus() {
        const endVal = document.getElementById('renew-new-end').value;
        if (!endVal) return;
        const endDate = new Date(endVal);
        const diff = Math.ceil((endDate - new Date()) / 86400000);
        document.getElementById('renew-status').value = diff < 0 ? 'Expired' : diff <= 90 ? 'Expiring Soon' : 'Active';
        // Compute validity in years+months from today
        const now = new Date();
        const yrs = endDate.getFullYear() - now.getFullYear();
        const mos = endDate.getMonth() - now.getMonth();
        const total = yrs * 12 + mos;
        const y = Math.floor(total / 12), m = total % 12;
        document.getElementById('renew-validity').value = (y > 0 ? y + ' yr' + (y > 1 ? 's' : '') : '') + (m > 0 ? (y > 0 ? ' ' : '') + m + ' mo' : '') || '< 1 month';
      }

        if (!newEnd) { alert('Please set a new end date.'); return; }
        const fmtDate = v => new Date(v).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
        const p = partnerships.find(x => x.id === renewingId);
        if (!p) return;
        p.end = fmtDate(newEnd);

        bootstrap.Modal.getInstance(document.getElementById('renewPartnershipModal'))?.hide();

        showToast(`"${p.inst}" renewed — new end date: ${p.end}.`);
      }

      // AUTO-COMPUTE STATUS FROM END DATE
      // ──────────────────────────────────────────────────────────
      function computeStatus() {
        const endVal = document.getElementById('f-end').value;
        if (!endVal) return;
        const today = new Date();
        const endDate = new Date(endVal);
        const diff = Math.ceil((endDate - today) / (1000 * 60 * 60 * 24));
        let status = 'Active';
        if (diff < 0) status = 'Expired';
        else if (diff <= 90) status = 'Expiring Soon';
        document.getElementById('f-status').value = status;
      }

      function submitPartnership() {
        const inst = document.getElementById('f-inst').value.trim();
        const type = document.getElementById('f-type').value;
        const start = document.getElementById('f-start').value;
        const end = document.getElementById('f-end').value;
        const unit = document.getElementById('f-unit').value;
        const region = document.getElementById('f-region').value;

        // Validation
        if (!inst) { highlight('f-inst', true); }
        if (!type) { highlight('f-type', true); }
        if (!start) { highlight('f-start', true); }
        if (!end) { highlight('f-end', true); }
        if (!unit) { highlight('f-unit', true); }
        if (!inst || !type || !start || !end || !unit) {
          return;
        }

        const fmtDate = v => new Date(v).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
        const diff = Math.ceil((new Date(end) - new Date()) / 86400000);
        const status = document.getElementById('f-status').value ||
          (diff < 0 ? 'Expired' : diff <= 90 ? 'Expiring Soon' : 'Active');

        const newP = {
          id: partnerships.length + 1,
          inst,
          country: document.getElementById('f-country').value.trim() || '—',
          region: region || '—',
          type,
          nature: document.getElementById('f-nature').value || '—',
          cat: document.getElementById('f-cat').value || 'International',
          unit,
          coordinator: document.getElementById('f-coordinator').value.trim(),
          partnerEmail: document.getElementById('f-partner-email').value.trim(),
          docLink: document.getElementById('f-doc-link').value.trim(),
          startYear: new Date(start).getFullYear(),
          endYear: new Date(end).getFullYear(),
          start: fmtDate(start),
          end: fmtDate(end),
          status,
          remarks: document.getElementById('f-remarks').value.trim(),
        };

        partnerships.push(newP);
        applyFilter();
        bootstrap.Modal.getInstance(document.getElementById('addPartnershipModal'))?.hide();
        showToast(`"${newP.inst}" added to the registry.`);

      }

      function searchPHInstitutions(q) {
        if (!q || q.length < 2) return [];
        const lower = q.toLowerCase();
        return PH_INSTITUTIONS
          .filter(i => i.name.toLowerCase().includes(lower) || i.city.toLowerCase().includes(lower))
          .slice(0, 8)
          .map(i => ({
            name: i.name,
            country: 'Philippines',
            domains: i.domain ? [i.domain] : [],
            city: i.city,
            'alpha_two_code': 'PH',
            _isLocal: true,
          }));
      }

      function countryToRegion(country) {
        return COUNTRY_REGION_MAP[country] || 'Local';
      }

      let acDebounce = null;
      let acResults = [];
      let acIndex = -1;

      function instSearch(val) {
        const dd = document.getElementById('ac-dropdown');
        const sp = document.getElementById('ac-spinner');
        acIndex = -1;

        if (!val || val.trim().length < 2) {
          dd.classList.remove('open');
          dd.innerHTML = '';
          return;
        }

        // ── 1. Show local PH results instantly (no API wait) ───────
        const localMatches = searchPHInstitutions(val.trim());
        if (localMatches.length) {
          acResults = localMatches;
          dd.innerHTML = renderAcItems(acResults);
          dd.classList.add('open');
        }

        // ── 2. Also fetch API results and merge ────────────────────
        clearTimeout(acDebounce);
        acDebounce = setTimeout(async () => {
          sp.style.display = 'block';
          try {
            const url = `/api/institutions?name=${encodeURIComponent(val.trim())}`;
            const res = await fetch(url);
            const data = await res.json();
            sp.style.display = 'none';

            // Deduplicate: skip names already in local results
            const existingNames = new Set(acResults.map(r => r.name.toLowerCase()));
            const apiResults = data.filter(u => !existingNames.has(u.name.toLowerCase())).slice(0, 10);

            // Merge: local PH first, then foreign API results
            acResults = [...localMatches, ...apiResults];

            if (!acResults.length) {
              dd.innerHTML = `<div class="ac-empty"><i class="fa-solid fa-magnifying-glass"></i> No results found for "<strong>${val}</strong>"</div>`;
              dd.classList.add('open');
              return;
            }

            dd.innerHTML = renderAcItems(acResults);
            dd.classList.add('open');
          } catch (e) {
            sp.style.display = 'none';
            // API failed — local PH results already shown, that's fine
            if (!acResults.length) {
              dd.innerHTML = `<div class="ac-empty"><i class="fa-solid fa-wifi-slash"></i> Could not reach API. Enter manually.</div>`;
              dd.classList.add('open');
            }
          }
        }, 350);
      }

      function renderAcItems(results) {
        return results.map((u, i) => {
          const badge = u._isLocal
            ? `<span class="ac-country" style="background:#dcfce7;color:#15803d;border:1px solid #bbf7d0;">🇵🇭 Local</span>`
            : `<span class="ac-country"><i class="fa-solid fa-location-dot fa-xs"></i> ${u.country}</span>`;
          const sub = u._isLocal && u.city
            ? `<span style="font-size:11px;color:#94a3b8;margin-left:4px;">· ${u.city}</span>`
            : '';
          return `<div class="ac-item" data-idx="${i}" onclick="instSelect(${i})">
        <span class="ac-name">${u.name}${sub}</span>
        ${badge}
      </div>`;
        }).join('');
      }

      function instSelect(idx) {
        const u = acResults[idx];
        if (!u) return;
        document.getElementById('f-inst').value = u.name;
        document.getElementById('f-country').value = u.country;
        // Auto-set Region
        const regionSel = document.getElementById('f-region');
        const region = countryToRegion(u.country);
        for (let i = 0; i < regionSel.options.length; i++) {
          if (regionSel.options[i].value === region) {
            regionSel.selectedIndex = i; break;
          }
        }
        // Auto-set Category
        const catSel = document.getElementById('f-cat');
        const isLocal = (u.country === 'Philippines');
        for (let i = 0; i < catSel.options.length; i++) {
          if (catSel.options[i].value === (isLocal ? 'Local' : 'International')) {
            catSel.selectedIndex = i; break;
          }
        }
        // Auto-suggest partner email from domain
        if (u.domains && u.domains.length) {
          const emailField = document.getElementById('f-partner-email');
          if (!emailField.value) emailField.value = `partnerships@${u.domains[0]}`;
        }
        // Close dropdown
        const dd = document.getElementById('ac-dropdown');
        dd.classList.remove('open');
        dd.innerHTML = '';
        acResults = [];
        acIndex = -1;
      }

      function instKeyNav(e) {
        const dd = document.getElementById('ac-dropdown');
        const items = dd.querySelectorAll('.ac-item');
        if (!items.length) return;
        if (e.key === 'ArrowDown') {
          e.preventDefault();
          acIndex = Math.min(acIndex + 1, items.length - 1);
          items.forEach((el, i) => el.classList.toggle('highlighted', i === acIndex));
          items[acIndex]?.scrollIntoView({ block: 'nearest' });
        } else if (e.key === 'ArrowUp') {
          e.preventDefault();
          acIndex = Math.max(acIndex - 1, 0);
          items.forEach((el, i) => el.classList.toggle('highlighted', i === acIndex));
          items[acIndex]?.scrollIntoView({ block: 'nearest' });
        } else if (e.key === 'Enter' && acIndex >= 0) {
          e.preventDefault();
          instSelect(acIndex);
        } else if (e.key === 'Escape') {
          dd.classList.remove('open');
          dd.innerHTML = '';
          acIndex = -1;
        }
      }

      // Close dropdown when clicking outside
      document.addEventListener('click', e => {
        if (!e.target.closest('.ac-wrapper')) {
          const dd = document.getElementById('ac-dropdown');
          if (dd) { dd.classList.remove('open'); dd.innerHTML = ''; }
          acIndex = -1;
        }
      });
